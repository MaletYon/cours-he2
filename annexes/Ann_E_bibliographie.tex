\chapter{Bibliographie Commentée}
\label{ann:bibliographie}

Pour chaque partie, quelques lectures, avec ce qu'elles apportent. Les
références complètes figurent dans la bibliographie du livre.

\section*{Partie I · Le cadre}
\begin{itemize}
    \item CIS Controls v8~\cite{cis2021controls}~: la place du test dans un programme de sécurité.
    \item NIST SP 800-115~\cite{nist800115}~: le cadre d'un test de sécurité.
    \item Lois 2010/012~\cite{loi2010012} et 2024/017~\cite{loi2024017}~: le droit applicable au Cameroun.
\end{itemize}

\section*{Partie II · Reconnaître}
\begin{itemize}
    \item OWASP WSTG~\cite{owaspwstg}~: la collecte d'informations sur une application.
    \item CVSS 3.1~\cite{first2019cvss31}, EPSS~\cite{firstepss}, catalogue KEV~\cite{cisakev}~: prioriser les vulnérabilités.
    \item Programmes CVE~\cite{mitrecve}~: identifier les vulnérabilités publiées.
\end{itemize}

\section*{Partie III · Les systèmes}
\begin{itemize}
    \item MITRE ATT\&CK~\cite{mitreattack}~: le vocabulaire des techniques.
    \item RFC 4120~\cite{rfc4120} et RFC 4795~\cite{rfc4795}~: Kerberos et la résolution locale.
    \item NIST SP 800-63B~\cite{nist80063b}~: les mots de passe.
\end{itemize}

\section*{Partie IV · Les applications}
\begin{itemize}
    \item OWASP Top 10~\cite{owasptop10} et API Top 10~\cite{owaspapitop10}~: les risques web et API.
    \item OWASP MASVS~\cite{owaspmasvs}~: la sécurité mobile.
    \item CIS Controls v8~\cite{cis2021controls}~: la configuration du nuage.
\end{itemize}

\section*{Partie V · Au-delà}
\begin{itemize}
    \item MITRE ATT\&CK~\cite{mitreattack}~: scénarios fondés sur une menace et couverture de détection.
\end{itemize}

\section*{Partie VI · Restituer}
\begin{itemize}
    \item PTES~\cite{ptes}~: la phase de restitution.
    \item ISO/IEC 29147~\cite{iso29147} et 30111~\cite{iso30111}~: la divulgation coordonnée.
    \item T.~E.~Minka Mi Nguidjoï~\cite{minka2026auditologie}~: le verdict à trois valeurs et la mesure.
\end{itemize}
